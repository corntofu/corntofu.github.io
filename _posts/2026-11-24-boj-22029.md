[문제 링크](https://acmicpc.net/problem/22029)

## Summary

투 스텝 문제다. N개의 정점을 가진 트리가 있을 때, 각 단계 때 다음과 같은 행위를 해야 한다.

1. $K(1 \leq k < \frac{N}{2})$개의 가짜 간선을 추가하고, 정점 하나에 표시를 한다.
2. 표시된 정점과 $N + K - 1$개의 간선을 보고 트리를 역으로 복원한다.

## Solution

일단 여러 가지 방법을 생각해 볼 수 있지만, 각 단계 별로 사고의 흐름을 서술하겠다.

### 1. 차수가 1인 정점에 마킹을 하고 그 정점에 간선을 연결

처음에는 이걸 떠올렸다. 차수가 1인, 즉 리프 노드인 정점 중 하나를 랜덤으로 잡아서 다른 1인 차수에 연결해서 차수를 2로 만드는 방법이었다.

하지만 1 - 2 - 3과 같은 트리 구조를 보면 구분이 불가능하기에 기각

### 2. 차수가 최대인 정점에 마킹을 하고 그 정점에서부터 거리가 같은 정점끼리 연결

차수가 최대인 정점을 X로 잡은 후, bfs로 거리값을 할당 후 같은 거릿값 끼리 간선을 추가하는 방법이다. 위의 1 - 2 - 3과 같은 경우에서 2에 마킹하고 1 - 3을 추가하면 구분할 수 있기에 1 - 2 - 3 트리 예제는 해결 가능하다.

하지만 이것도 문제점이 있는데, 1 - 2 - 3 - ... -N과 같은 경우를 생각해보자. $1$과 $N$을 제외하면 모두 차수가 2인데, 어디 할당해야 할까? 만약 2에 할당하면 1 - 3에 간선을 추가하는 것 말고는 할 수 있는게 없을 것이다. 하지만 중앙에 할당하면 $K$를 다 채울 수 있다. 그렇다면 차수가 최대인 게 아니라 중앙에 마킹을 하면 되지 않을까?

### 3. 트리의 중심에 마킹을 하고 그 정점에서부터 거리가 같은 정점끼리 연결

아까에서 조금만 바뀌었다. 트리의 중심을 $C$라고 하고, $C$에서의 거리가 $d$인 정점의 개수를 $c_d$, 중심에서 가장 먼 정점의 거리를 $D$라 할 때, 간선의 개수는 $\sum \limits_{d = 1}^D \binom{c_d}{2}$다. 참고로 이항계수 값은 $c_d > 0$에서 $c_d - 1$보다 동일하거나 큰데, $\sum \limits_{d = 1}^D \binom{c_d}{2} \geq \sum \limits_{d = 1}^D (c_d - 1) = (N - 1) - D$이다. 트리의 지름이기에 $D \leq \lfloor \frac{N}{2} \rfloor$이고 $N - D - 1 = N - \lfloor \frac{N}{2} \rfloor - 1 = \lceil \frac{N}{2} \rceil - 1 \geq K$. 


즉 모든 $K$에 대해서 간선 개수를 만족한다. 

## Code

```cpp
#include "railroad.h"
#include <bits/stdc++.h>
using namespace std;
using pii = pair<int, int>;

vector<int> edge[202];
vector<int> d[202];

int center_of_tree(int N) {
    auto bfs = [&](int start) {
        vector<int> dist(N + 1, -1);
        vector<int> parent(N + 1, -1);
        queue<int> q;
        q.push(start);
        dist[start] = 0;
        int farthest = start;

        while (!q.empty()) {
            int u = q.front(); q.pop();
            if (dist[u] > dist[farthest]) farthest = u;
            for (int v : edge[u]) {
                if (dist[v] == -1) {
                    dist[v] = dist[u] + 1;
                    parent[v] = u;
                    q.push(v);
                }
            }
        }
        return make_pair(farthest, parent);
    };

    int A = bfs(1).first;
    auto [B, parent] = bfs(A);

    vector<int> path;
    for (int cur = B; cur != -1; cur = parent[cur]) {
        path.push_back(cur);
    }

    return path[path.size() / 2];
}

void bfs(int start, int N) {
    vector<int> dist(N + 1, -1);
    queue<int> q;
    q.push(start);
    dist[start] = 0;

    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : edge[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                d[dist[v]].push_back(v);
                q.push(v);
            }
        }
    }
}

vector<pair<int, int>> encode_map(int N, int K, int &X, vector<pair<int, int>> E) {
    for (int i = 1; i <= N; i++) {
        edge[i].clear();
        d[i].clear();
    }

    for (auto [a, b] : E) {
        edge[a].emplace_back(b);
        edge[b].emplace_back(a);
    }

    X = center_of_tree(N);
    bfs(X, N);

    vector<pii> ret;
    for (int i = 1; i <= N; i++) {
        for (int j = 0; j < d[i].size(); j++) {
            for (int k = j + 1; k < d[i].size(); k++) {
                ret.emplace_back(d[i][j], d[i][k]);
                if ((int)ret.size() == K) return ret;
            }
        }
    }

    return ret;
}

vector<pair<int, int>> decode_map(int N, int K, int X, vector<pair<int, int>> E) {
    vector<vector<int>> adj(N + 1);
    for (auto [a, b] : E) {
        adj[a].push_back(b);
        adj[b].push_back(a);
    }

    vector<int> dist(N + 1, -1);
    queue<int> q;
    q.push(X);
    dist[X] = 0;

    while (!q.empty()) {
        int u = q.front(); q.pop();
        for (int v : adj[u]) {
            if (dist[v] == -1) {
                dist[v] = dist[u] + 1;
                q.push(v);
            }
        }
    }

    vector<pii> ret;
    for (auto [a, b] : E) {
        if (dist[a] != dist[b]) {
            ret.emplace_back(a, b);
        }
    }

    return ret;
}
```
